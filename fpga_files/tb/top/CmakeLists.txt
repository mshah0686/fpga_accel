# 1. Define the simulation executable binary name and its C++ testbench source
set(TB_NAME top_tb)
add_executable(${TB_NAME} tb_top.cpp)

# 2. Compile the full-chip Verilog hardware files using the official Verilator macro
verilate(${TB_NAME}
    PREFIX Vtb_top
    TOP_MODULE tb_top
    TRACE_VCD
    INCLUDE_DIRS
        "${CMAKE_SOURCE_DIR}/rtl"
        "${CMAKE_SOURCE_DIR}/rtl/spi"
        "${CMAKE_SOURCE_DIR}/rtl/compute"
    VERILATOR_ARGS
        "-timescale"
        "1ns/1ps"
        "--timing"
        "--trace"
        "--trace-structs"
    SOURCES
        "${CMAKE_SOURCE_DIR}/rtl/fpga_types.sv"

        # SPI front-end (RX + CDC + decode + TX FIFO)
        "${CMAKE_SOURCE_DIR}/rtl/spi/spi_rx.sv"
        "${CMAKE_SOURCE_DIR}/rtl/spi/spi_rx_sync.sv"
        "${CMAKE_SOURCE_DIR}/rtl/spi/spi_tx.sv"
        "${CMAKE_SOURCE_DIR}/rtl/spi/sync_flops.sv"
        "${CMAKE_SOURCE_DIR}/rtl/spi/bin2gray.sv"
        "${CMAKE_SOURCE_DIR}/rtl/spi/fifo_async.sv"
        "${CMAKE_SOURCE_DIR}/rtl/spi/packet_decoder.sv"
        "${CMAKE_SOURCE_DIR}/rtl/spi_top.sv"

        # Register model + peripherals
        "${CMAKE_SOURCE_DIR}/rtl/register_model.sv"
        "${CMAKE_SOURCE_DIR}/rtl/timer.sv"
        "${CMAKE_SOURCE_DIR}/rtl/segment_display.sv"

        # Matrix multiplier: BRAM storage + wrapper feed the systolic array
        "${CMAKE_SOURCE_DIR}/rtl/compute/single_port_bram.sv"
        "${CMAKE_SOURCE_DIR}/rtl/compute/bram_wrapper.sv"
        "${CMAKE_SOURCE_DIR}/rtl/compute/mac.sv"
        "${CMAKE_SOURCE_DIR}/rtl/compute/matrix_mult_datapath.sv"
        "${CMAKE_SOURCE_DIR}/rtl/compute/matrix_mult_controller.sv"
        "${CMAKE_SOURCE_DIR}/rtl/compute/matrix_mult_top.sv"

        # Chip top + testbench
        "${CMAKE_SOURCE_DIR}/rtl/top.sv"
        "${CMAKE_CURRENT_SOURCE_DIR}/tb_top.v"
)

# 3. Explicitly link the core Verilator static library via its absolute path variable
target_link_libraries(${TB_NAME}
    PRIVATE
        ${VERILATOR_LIBS}
)

# 4. single_port_bram's $readmemh uses paths like "weights/hidden_0.hex",
#    resolved against the simulator's working directory (fpga_files/), while
#    the real files live in fpga_files/rtl/weights. Symlink them into place.
if(NOT EXISTS "${CMAKE_SOURCE_DIR}/weights")
    file(CREATE_LINK
        "${CMAKE_SOURCE_DIR}/rtl/weights"
        "${CMAKE_SOURCE_DIR}/weights"
        SYMBOLIC)
endif()

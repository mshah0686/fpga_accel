# 1. Define the simulation executable binary name and its C++ testbench source
set(TB_NAME top_tb)
add_executable(${TB_NAME} tb_top.cpp)

# 2. Compile the full-chip Verilog hardware files using the official Verilator macro
verilate(${TB_NAME}
    PREFIX Vtb_top
    TOP_MODULE tb_top
    TRACE_VCD
    INCLUDE_DIRS
        "${CMAKE_SOURCE_DIR}/rtl"
        "${CMAKE_SOURCE_DIR}/rtl/spi"
        "${CMAKE_SOURCE_DIR}/rtl/compute"
    VERILATOR_ARGS
        "-timescale"
        "1ns/1ps"
        "--timing"
        "--trace"
        "--trace-structs"
    SOURCES
        "${CMAKE_SOURCE_DIR}/rtl/fpga_types.v"

        # SPI front-end (RX + CDC + decode + TX FIFO)
        "${CMAKE_SOURCE_DIR}/rtl/spi/spi_rx.v"
        "${CMAKE_SOURCE_DIR}/rtl/spi/spi_rx_sync.v"
        "${CMAKE_SOURCE_DIR}/rtl/spi/spi_tx.v"
        "${CMAKE_SOURCE_DIR}/rtl/spi/sync_flops.v"
        "${CMAKE_SOURCE_DIR}/rtl/spi/bin2gray.v"
        "${CMAKE_SOURCE_DIR}/rtl/spi/fifo_async.v"
        "${CMAKE_SOURCE_DIR}/rtl/spi/packet_decoder.v"
        "${CMAKE_SOURCE_DIR}/rtl/spi_top.v"

        # Register model + peripherals
        "${CMAKE_SOURCE_DIR}/rtl/register_model.v"
        "${CMAKE_SOURCE_DIR}/rtl/timer.v"
        "${CMAKE_SOURCE_DIR}/rtl/segment_display.v"

        # 2x2 matrix multiplier
        "${CMAKE_SOURCE_DIR}/rtl/compute/mac.v"
        "${CMAKE_SOURCE_DIR}/rtl/compute/matrix_mult_datapath.v"
        "${CMAKE_SOURCE_DIR}/rtl/compute/matrix_mult_controller.v"
        "${CMAKE_SOURCE_DIR}/rtl/compute/matrix_mult_top.v"

        # Chip top + testbench
        "${CMAKE_SOURCE_DIR}/rtl/top.v"
        "${CMAKE_CURRENT_SOURCE_DIR}/tb_top.v"
)

# 3. Explicitly link the core Verilator static library via its absolute path variable
target_link_libraries(${TB_NAME}
    PRIVATE
        ${VERILATOR_LIBS}
)

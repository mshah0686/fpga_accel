# 1. Define the simulation executable binary name and its C++ testbench source
set(TB_NAME matrix_mult_tb)
add_executable(${TB_NAME} tb_matrix_mult.cpp)

# 2. Compile your Verilog hardware files using the official Verilator macro
verilate(${TB_NAME}
    PREFIX Vtb_matrix_mult
    TOP_MODULE tb_matrix_mult_top
    TRACE_VCD
    INCLUDE_DIRS
        "${CMAKE_SOURCE_DIR}/rtl"
    VERILATOR_ARGS
        "-timescale"
        "1ns/1ps"
        "--timing"
        "--trace"
        "--trace-structs"
    SOURCES
        "${CMAKE_SOURCE_DIR}/rtl/compute/mac.sv"
        "${CMAKE_SOURCE_DIR}/rtl/compute/matrix_mult_datapath.sv"
        "${CMAKE_SOURCE_DIR}/rtl/compute/matrix_mult_controller.sv"
        "${CMAKE_SOURCE_DIR}/rtl/compute/matrix_mult_top.sv"
        "${CMAKE_CURRENT_SOURCE_DIR}/tb_matrix_mult.v"
)

# 3. Explicitly link the core Verilator static library via its absolute path variable
target_link_libraries(${TB_NAME}
    PRIVATE
        ${VERILATOR_LIBS}
)

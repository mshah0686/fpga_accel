# 1. Define the simulation executable binary name and its C++ testbench source
set(TB_NAME matrix_mult_tb)
add_executable(${TB_NAME} tb_matrix_mult.cpp)

# 2. Compile your Verilog hardware files using the official Verilator macro
verilate(${TB_NAME}
    PREFIX Vtb_matrix_mult
    TOP_MODULE tb_matrix_mult_top
    TRACE_VCD
    INCLUDE_DIRS
        "${CMAKE_SOURCE_DIR}/rtl"
        "${CMAKE_SOURCE_DIR}/rtl/compute"
        "${CMAKE_CURRENT_SOURCE_DIR}"
    VERILATOR_ARGS
        "-timescale"
        "1ns/1ps"
        "--timing"
        "--trace"
        "--trace-structs"
    SOURCES
        # BRAM storage + per-edge BRAM array wrapper (operands now live here,
        # preloaded from rtl/weights/*.hex via $readmemh)
        "${CMAKE_SOURCE_DIR}/rtl/compute/single_port_bram.sv"
        "${CMAKE_SOURCE_DIR}/rtl/compute/bram_wrapper.sv"

        # Systolic PE + datapath + FSM + wrapper (no package/include deps)
        "${CMAKE_SOURCE_DIR}/rtl/compute/mac.sv"
        "${CMAKE_SOURCE_DIR}/rtl/compute/matrix_mult_datapath.sv"
        "${CMAKE_SOURCE_DIR}/rtl/compute/matrix_mult_controller.sv"
        "${CMAKE_SOURCE_DIR}/rtl/compute/matrix_mult_top.sv"

        # Testbench (16x784 * 784x1, DATA_WIDTH=8 / ACC_WIDTH=26).
        # Drives only the control-register req pulse; operands come from BRAM.
        "${CMAKE_CURRENT_SOURCE_DIR}/tb_matrix_mult.v"
)

# 3. Explicitly link the core Verilator static library via its absolute path variable
target_link_libraries(${TB_NAME}
    PRIVATE
        ${VERILATOR_LIBS}
)

# 4. The C++ TB opens waves/tb_matrix_mult.vcd relative to the run directory
#    (run the binary from the fpga_files/ root, same as top_tb).
file(MAKE_DIRECTORY "${CMAKE_SOURCE_DIR}/waves")

# 5. single_port_bram's $readmemh is called with paths like "weights/hidden_0.hex"
#    (see LOAD_FILE_PREFIX in matrix_mult_top). $readmemh resolves relative
#    paths against the simulator's working directory, which is fpga_files/,
#    but the real files live in fpga_files/rtl/weights. Drop a symlink at the
#    expected location so the preload finds them without touching the RTL.
if(NOT EXISTS "${CMAKE_SOURCE_DIR}/weights")
    file(CREATE_LINK
        "${CMAKE_SOURCE_DIR}/rtl/weights"
        "${CMAKE_SOURCE_DIR}/weights"
        SYMBOLIC)
endif()
